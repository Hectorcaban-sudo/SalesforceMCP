# Use the standard Windows Server Core 2022 base image
FROM mcr.microsoft.com/windows/servercore:ltsc2022

# Set PowerShell as the default shell for running commands
SHELL ["powershell", "-Command", "$ErrorActionPreference = 'Stop'; $ProgressPreference = 'SilentlyContinue';"]

# The download URL for the standalone Salesforce CLI
ENV SF_CLI_URL="https://developer.salesforce.com/media/salesforce-cli/sf/channels/stable/sf-win32-x64.tar.gz"

# Download, extract, clean up, and set the system PATH in a single RUN command
# Doing this in one step keeps the Docker image size as small as possible
RUN Write-Host 'Downloading Salesforce CLI...' ; \
    Invoke-WebRequest -Uri $ENV:SF_CLI_URL -OutFile C:\sf-cli.tar.gz ; \
    \
    Write-Host 'Extracting to C:\sf_cli...' ; \
    New-Item -ItemType Directory -Path C:\sf_cli | Out-Null ; \
    tar -xzf C:\sf-cli.tar.gz -C C:\sf_cli --strip-components 1 ; \
    \
    Write-Host 'Cleaning up downloaded archive...' ; \
    Remove-Item C:\sf-cli.tar.gz ; \
    \
    Write-Host 'Adding Salesforce CLI to Machine PATH...' ; \
    $newPath = 'C:\sf_cli\bin;' + [Environment]::GetEnvironmentVariable('PATH', [EnvironmentVariableTarget]::Machine) ; \
    [Environment]::SetEnvironmentVariable('PATH', $newPath, [EnvironmentVariableTarget]::Machine)

# Set the working directory
WORKDIR C:\app

# Set the default command to verify the installation
CMD ["sf.cmd", "version"]
