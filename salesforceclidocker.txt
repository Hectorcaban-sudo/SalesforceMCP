# ==========================================
# Stage 1: Build & Extraction (Server Core)
# ==========================================
FROM mcr.microsoft.com/windows/servercore:ltsc2022 AS builder

# Use PowerShell for the build stage
SHELL ["powershell", "-Command", "$ErrorActionPreference = 'Stop'; $ProgressPreference = 'SilentlyContinue';"]

# Set the URL for the Windows x64 Salesforce CLI tarball
ENV SF_CLI_URL="https://developer.salesforce.com/media/salesforce-cli/sf/channels/stable/sf-win32-x64.tar.gz"

RUN New-Item -ItemType Directory -Path C:\downloads; \
    New-Item -ItemType Directory -Path C:\sf_cli; \
    Write-Host 'Downloading Salesforce CLI...'; \
    Invoke-WebRequest -Uri $ENV:SF_CLI_URL -OutFile C:\downloads\sf-cli.tar.gz; \
    Write-Host 'Extracting tarball...'; \
    tar -xzf C:\downloads\sf-cli.tar.gz -C C:\sf_cli --strip-components 1

# ==========================================
# Stage 2: Final Runtime Image (Nano Server)
# ==========================================
FROM mcr.microsoft.com/windows/nanoserver:ltsc2022

# Switch to ContainerAdministrator to modify the system PATH persistently
USER ContainerAdministrator

# Copy the extracted Salesforce CLI directory from the builder stage
COPY --from=builder C:\sf_cli C:\sf_cli

# Add the Salesforce CLI bin directory to the system PATH
RUN setx /M PATH "%PATH%;C:\sf_cli\bin"

# Switch back to the secure, default user
USER ContainerUser

WORKDIR C:\app

# Set the default command to verify the installation
CMD ["sf.cmd", "version"]
